cmake_minimum_required(VERSION 3.20)
project(TemplatesAndMetaprogramming LANGUAGES CXX)

# Устанавливаем стандарт C++
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF) # выключаем MSVC расширения, чтобы код был максимально переносим


# Добавляем исполняемый target
add_executable(templateFactorial
    "${CMAKE_CURRENT_SOURCE_DIR}/templateFactorial/cpp/templateFactorial.cpp"
)
add_executable(templateTypeList
    "${CMAKE_CURRENT_SOURCE_DIR}/templateTypeList/cpp/templateTypeList.cpp"
)
add_executable(templatePrint
    "${CMAKE_CURRENT_SOURCE_DIR}/templatePrint/cpp/templatePrint.cpp"
)
add_executable(templateAdd
    "${CMAKE_CURRENT_SOURCE_DIR}/templateAdd/cpp/templateAdd.cpp"
)
add_executable(templateMyTypeTraits
    "${CMAKE_CURRENT_SOURCE_DIR}/templateMyTypeTraits/cpp/templateMyTypeTraits.cpp"
)
add_executable(templatePolicyVector
    "${CMAKE_CURRENT_SOURCE_DIR}/templatePolicyVector/cpp/templatePolicyVector.cpp"
)

# Указываем, где лежат наши .hpp
target_include_directories(templateFactorial
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/templateFactorial"
)
target_include_directories(templateTypeList
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/templateTypeList"
)
target_include_directories(templatePrint
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/templatePrint"
)
target_include_directories(templateAdd
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/templateAdd"
)
target_include_directories(templateMyTypeTraits
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/templateMyTypeTraits"
)
target_include_directories(templatePolicyVector
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/templatePolicyVector"
)
